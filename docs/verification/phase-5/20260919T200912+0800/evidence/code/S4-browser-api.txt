    34		value, err := b.dispatch(ctx, method, endpoint, session, body)
    35		if err != nil {

    51	func (s *Server) Handler() http.Handler {
    52		return http.HandlerFunc(func(response http.ResponseWriter, request *http.Request) {
    53			corsHeaders(response, request, s.trustedOrigin)
    54			wrapped := &commonResponseWriter{ResponseWriter: response, secure: s.secureCookies, request: request}
    55			if request.Method == http.MethodOptions {
    56				optionsResponse(wrapped, request, s.trustedOrigin)
    57				return
    58			}
    59			s.route(wrapped, request)
    60		})

    63	func (s *Server) route(response http.ResponseWriter, request *http.Request) {
    64		path := request.URL.Path
    65	
    66		if path == "/health/live" {
    67			if !method(response, request, http.MethodGet) {
    68				return
    69			}
    70			writeJSON(response, http.StatusOK, liveHealth())
    71			return
    72		}
    73		if path == "/health/ready" {
    74			if !method(response, request, http.MethodGet) {
    75				return
    76			}
    77			if !isReady(request.Context(), s.backend.Health) {
    78				writeJSON(response, http.StatusServiceUnavailable, unavailableHealth())
    79				return
    80			}
    81			writeJSON(response, http.StatusOK, readyHealth())
    82			return
    83		}
    84		if path == "/api/platform/ping" {
    85			if !method(response, request, http.MethodGet) {
    86				return
    87			}
    88			value, err := s.backend.DoJSON(request.Context(), http.MethodGet, "/internal/platform/ping", "", nil)
    89			if err != nil {
    90				writeError(response, http.StatusBadGateway, "core_unavailable", "Core platform is unavailable")
    91				return
    92			}
    93			writeJSON(response, http.StatusOK, value)
    94			return
    95		}
    96	
    97		// Authentication has a small number of anonymous endpoints.
    98		if path == "/auth/session" {
    99			if !method(response, request, http.MethodGet) {
   100				return
   101			}
   102			value, err := s.backend.DoJSON(request.Context(), http.MethodGet, "/internal/auth/session", cookieValue(request, sessionCookieName), nil)
   103			if err != nil || !truthy(value["authenticated"]) {
   104				clearSessionCookie(response, s.secureCookies)
   105				writeJSON(response, http.StatusUnauthorized, map[string]any{"authenticated": false})
   106				return
   107			}
   108			sessionResponse := map[string]any{"authenticated": true}
   109			if actorID := first(value, "actorId", "actor_id"); actorID != nil {
   110				sessionResponse["actorId"] = actorID
   111			}
   112			writeJSON(response, http.StatusOK, sessionResponse)
   113			return
   114		}
   115		if path == "/auth/setup-status" {
   116			if !method(response, request, http.MethodGet) {
   117				return
   118			}
   119			value, err := s.backend.DoJSON(request.Context(), http.MethodGet, "/internal/auth/setup-status", "", nil)
   120			if err != nil {
   121				writeError(response, http.StatusBadGateway, "core_unavailable", "Core authentication is unavailable")
   122				return
   123			}
   124			writeJSON(response, http.StatusOK, map[string]any{"setupAvailable": truthy(first(value, "setup_available", "setupAvailable"))})
   125			return
   126		}
   127		if path == "/auth/login" || path == "/auth/setup" {
   128			if !method(response, request, http.MethodPost) {
   129				return
   130			}
   131			body, ok := readBody(response, request)
   132			if !ok || !validatePassword(body["password"]) || (path == "/auth/setup" && !validateString(body["setupToken"], 16, 1<<20)) {
   133				if ok {
   134					writeError(response, http.StatusBadRequest, "invalid_request", "Request validation failed")
   135				}
   136				return
   137			}
   138			if !mutationGuard(response, request, s.trustedOrigin) {
   139				return
   140			}
   141			coreBody := map[string]any{"password": body["password"]}
   142			endpoint := "/internal/auth/login"
   143			if path == "/auth/setup" {
   144				endpoint = "/internal/auth/setup"
   145				coreBody["setup_token"] = body["setupToken"]
   146			}
   147			value, err := s.backend.DoJSON(request.Context(), http.MethodPost, endpoint, "", coreBody)
   148			if err != nil {
   149				writeJSON(response, http.StatusUnauthorized, map[string]any{"authenticated": false})
   150				return
   151			}
   152			sessionToken := stringValue(first(value, "session_token", "sessionToken"))
   153			if sessionToken == "" || !truthy(value["authenticated"]) {
   154				writeJSON(response, http.StatusUnauthorized, map[string]any{"authenticated": false})
   155				return
   156			}
   157			setSessionCookie(response, sessionToken, s.secureCookies)
   158			setCSRFCookie(response, newCSRFToken(), s.secureCookies)
   159			authenticated := map[string]any{"authenticated": true}
   160			if actorID := first(value, "actor_id", "actorId"); actorID != nil {
   161				authenticated["actorId"] = actorID
   162			}
   163			writeJSON(response, http.StatusOK, authenticated)
   164			return
   165		}
   166		if path == "/auth/logout" || path == "/auth/revoke-all" || path == "/auth/password" {
   167			if !method(response, request, http.MethodPost) {
   168				return
   169			}
   170			var passwordBody map[string]any
   171			if path == "/auth/password" {
   172				var valid bool
   173				passwordBody, valid = readBody(response, request)
   174				if !valid || !validatePassword(passwordBody["password"]) {
   175					if valid {
   176						writeError(response, http.StatusBadRequest, "invalid_request", "Request validation failed")
   177					}
   178					return
   179				}
   180			}
   181			if !mutationGuard(response, request, s.trustedOrigin) {
   182				return
   183			}
   184			session := cookieValue(request, sessionCookieName)
   185			if path != "/auth/logout" && session == "" {
   186				writeError(response, http.StatusUnauthorized, "unauthenticated", "Authentication is required")
   187				return
   188			}
   189			if path == "/auth/password" {
   190				if _, err := s.backend.DoJSON(request.Context(), http.MethodPost, "/internal/auth/reset-password", session, map[string]any{"password": passwordBody["password"]}); err != nil {
   191					writeError(response, http.StatusForbidden, "password_change_failed", "Password could not be changed")
   192					return
   193				}
   194				clearSessionCookie(response, s.secureCookies)
   195				setCSRFCookie(response, newCSRFToken(), s.secureCookies)
   196				response.WriteHeader(http.StatusNoContent)
   197				return
   198			}
   199			if session != "" {
   200				endpoint := "/internal/auth/revoke-current"
   201				failureCode, failureMessage := "logout_failed", "Session could not be revoked"
   202				if path == "/auth/revoke-all" {
   203					endpoint, failureCode, failureMessage = "/internal/auth/revoke-all", "revoke_failed", "Session revocation failed"
   204				}
   205				if _, err := s.backend.DoJSON(request.Context(), http.MethodPost, endpoint, session, nil); err != nil {
   206					if path == "/auth/logout" {
   207						clearSessionCookie(response, s.secureCookies)
   208						setCSRFCookie(response, newCSRFToken(), s.secureCookies)
   209						response.WriteHeader(http.StatusNoContent)
   210						return
   211					}
   212					writeError(response, http.StatusForbidden, failureCode, failureMessage)
   213					return
   214				}
   215			}
   216			clearSessionCookie(response, s.secureCookies)
   217			setCSRFCookie(response, newCSRFToken(), s.secureCookies)
   218			response.WriteHeader(http.StatusNoContent)
   219			return
