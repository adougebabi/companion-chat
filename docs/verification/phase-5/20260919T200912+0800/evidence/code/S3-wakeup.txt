   372	// RunADKStructuredTask uses the shared surface-aware ADK bridge. It does not
   373	// freeze, settle, publish or open a transaction; the caller owns those rules.
   374	func (a *App) RunADKStructuredTask(ctx context.Context, input ADKStructuredTaskInput) (ADKStructuredTaskResult, error) {
   375		if a == nil || a.Provider == nil {
   376			return ADKStructuredTaskResult{}, errors.New("adk_structured_task_provider_unavailable")
   377		}
   378		if !isADKLoopSchema(input.SchemaName) {
   379			return ADKStructuredTaskResult{}, fmt.Errorf("adk_schema_not_allowed: %s", strings.TrimSpace(input.SchemaName))
   380		}
   381		if len(input.Definitions) > 0 && input.Capability == nil {
   382			return ADKStructuredTaskResult{}, errors.New("adk_structured_task_capability_context_required")
   383		}
   384		if err := validateADKCapabilityDefinitions(a, input.Definitions, firstCapabilitySurface(input.Capability)); err != nil {
   385			return ADKStructuredTaskResult{}, err
   386		}
   387		trace := &ADKCapabilityTrace{}
   388		if input.Capability != nil {
   389			invoker := newAppADKCapabilityInvoker(a, *input.Capability, trace)
   390			ctx = WithADKCapabilityInvoker(ctx, invoker, trace)
   391		}
   392		if strings.TrimSpace(input.Scenario) != "" {
   393			ctx = WithProviderScenario(ctx, input.Scenario)
   394		}
   395		completion, err := a.Provider.StructuredAssembledWithToolsSchema(ctx, input.Role, input.Messages, input.Definitions, input.SchemaName, input.Schema, input.EnableThinking)
   396		if err != nil {
   397			return ADKStructuredTaskResult{Trace: trace}, err
   398		}
   399		return ADKStructuredTaskResult{Completion: completion, Trace: trace}, nil
   400	}
