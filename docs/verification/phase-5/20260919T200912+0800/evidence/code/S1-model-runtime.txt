   137	func (m *queuedToolCallingChatModel) Generate(ctx context.Context, input []*schema.Message, opts ...model.Option) (*schema.Message, error) {
   138		input = normalizeEinoToolMessageNames(input)
   139		sequence := uint64(1)
   140		if m.sequence != nil {
   141			sequence = m.sequence.Add(1)
   142		}
   143		callCorrelation := fmt.Sprintf("%s:adk:%d", m.correlationID, sequence)
   144		callRequestID := providerDiagnosticRequestID(m.role, callCorrelation)
   145		callCtx := withEinoRequestID(withoutProviderQueueBypass(ctx), callRequestID)
   146		// The parent turn attempt is intentionally not reused as the physical
   147		// model-call attempt. Diagnostics and cancellation state must distinguish
   148		// Generate #1 from Generate #2 while both share one turn correlation.
   149		callCtx = WithProviderAttemptIdentity(callCtx, randomID("provider_attempt_"))
   150		callDiagnosticID := ""
   151		if m.provider != nil && m.provider.DB != nil {
   152			callDiagnosticID = (&App{DB: m.provider.DB}).recordQueuedModelRun(callCtx, m.role, m.assignment.EndpointID, m.assignment.ModelID, callCorrelation, m.scenario, m.priority, einoDiagnosticMessages(input))
   153		}
   154		started := time.Now()
   155		result, err := runProviderQueued(m.provider, callCtx, m.role, m.scenario, m.priority, callDiagnosticID, func(runCtx context.Context) (*schema.Message, error) {
   156			return m.inner.Generate(runCtx, input, opts...)
   157		})
   158		if callDiagnosticID != "" {
   159			(&App{DB: m.provider.DB}).updateModelRunPromptMetrics(callCtx, callDiagnosticID, einoUsage(result), time.Since(started))
   160		}
   161		return result, err
   162	}

   201	func einoDiagnosticMessages(messages []*schema.Message) []map[string]any {
   202		result := make([]map[string]any, 0, len(messages))
   203		for _, message := range messages {
   204			if message == nil {
   205				continue
   206			}
   207			result = append(result, einoMessageRaw(message))
   208			if len(result) >= 64 {
   209				break
   210			}
   211		}
   212		return result
   213	}

   273	func (p *ProviderClient) refreshQueueLimits(ctx context.Context) {
   274		generated, embedding := providerQueueDefaultConcurrency, providerQueueDefaultEmbedding
   275		if p != nil && p.DB != nil {
   276			var raw string
   277			if err := p.DB.Pool().QueryRow(ctx, `SELECT value_json FROM public.runtime_settings WHERE key='llm.queue'`).Scan(&raw); err == nil {
   278				var settings providerQueueSettings
   279				if json.Unmarshal([]byte(raw), &settings) == nil {
   280					if settings.GeneratedConcurrency != 0 {
   281						generated = settings.GeneratedConcurrency
   282					}
   283					if settings.EmbeddingConcurrency != 0 {
   284						embedding = settings.EmbeddingConcurrency
   285					}
   286				}
   287			}
   288		}
   289		p.queueFor("generic_llm").setLimit(generated)
   290		p.queueFor("embedding").setLimit(embedding)

   293	func runProviderQueued[T any](p *ProviderClient, ctx context.Context, role, scenario string, priority int, diagnosticID string, fn func(context.Context) (T, error)) (T, error) {
   294		var result T
   295		if p == nil {
   296			return result, errors.New("provider_unavailable")
   297		}
   298		if providerQueueBypassed(ctx) {
   299			return fn(ctx)
   300		}
   301		p.refreshQueueLimits(ctx)
   302		runCtx, cancel := context.WithCancel(ctx)
   303		defer cancel()
   304		stopWatch := p.watchProviderCancellation(runCtx, cancel)
   305		defer stopWatch()
   306		queue := p.queueFor(role)
   307		releaseRedis, redisEnabled, redisErr := p.acquireProviderRedisSlot(runCtx, role, priority, queue.currentLimit(), diagnosticID)
   308		if redisErr != nil {
   309			if diagnosticID != "" {
   310				(&App{DB: p.DB}).updateModelRunState(ctx, diagnosticID, providerRunFailed, redisErr)
   311			}
   312			return result, redisErr
   313		}
   314		if redisEnabled {
   315			defer releaseRedis()
   316		}
   317		err := queue.submit(runCtx, priority, func(taskCtx context.Context) error {
   318			if guard := providerExecutionGuard(taskCtx); guard != nil {
   319				if guardErr := guard(taskCtx); guardErr != nil {
   320					return guardErr
   321				}
   322			}
   323			var runErr error
   324			result, runErr = fn(taskCtx)
   325			return runErr
   326		}, func(status string, runErr error) {
   327			if p.DB == nil || diagnosticID == "" {
   328				return
   329			}
   330			(&App{DB: p.DB}).updateModelRunState(ctx, diagnosticID, status, runErr)
   331		})
   332		return result, err
   333	}
