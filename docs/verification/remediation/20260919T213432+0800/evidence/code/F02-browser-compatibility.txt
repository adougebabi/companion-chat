   165	// jsonValue preserves diagnostic JSON values such as the provider message
   166	// array. objectValue is intentionally stricter for DTOs that require an
   167	// object, but Prompt/Response are documented as arbitrary JSON payloads.
   168	func jsonValue(value any) any {
   169		if value == nil {
   170			return map[string]any{}
   171		}
   172		switch typed := value.(type) {
   173		case json.RawMessage:
   174			var decoded any
   175			if json.Unmarshal(typed, &decoded) == nil && decoded != nil {
   176				return decoded
   177			}
   178		case []byte:
   179			var decoded any
   180			if json.Unmarshal(typed, &decoded) == nil && decoded != nil {
   181				return decoded
   182			}
   183		}
   184		return value
   185	}
