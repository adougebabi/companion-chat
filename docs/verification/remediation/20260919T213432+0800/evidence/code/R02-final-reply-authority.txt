   254	// RunADKLoop is the only Eino ADK model/tool event iterator used by Core.
   255	// Callers receive bounded messages and hand the structured result to their
   256	// existing domain freeze/settlement boundary.
   257	func RunADKLoop(ctx context.Context, config ADKLoopConfig, messages []*schema.Message) (ADKLoopResult, error) {
   258		if config.Model == nil {
   259			return ADKLoopResult{}, errors.New("adk_model_required")
   260		}
   261		if strings.TrimSpace(config.Name) == "" || strings.TrimSpace(config.Description) == "" {
   262			return ADKLoopResult{}, errors.New("adk_agent_identity_required")
   263		}
   264		if len(messages) == 0 {
   265			return ADKLoopResult{}, errors.New("adk_messages_required")
   266		}
   267		maxIterations := config.MaxIterations
   268		if maxIterations <= 0 {
   269			maxIterations = 2
   270		}
   271		if maxIterations > 2 {
   272			return ADKLoopResult{}, errors.New("adk_iteration_limit_invalid")
   273		}
   274		agent, err := adk.NewChatModelAgent(ctx, &adk.ChatModelAgentConfig{
   275			Name: config.Name, Description: config.Description,
   276			Instruction: config.Instruction, Model: config.Model,
   277			ToolsConfig: adk.ToolsConfig{ToolsNodeConfig: compose.ToolsNodeConfig{
   278				Tools: config.Tools, ExecuteSequentially: true,
   279			}},
   280			MaxIterations: maxIterations,
   281		})
   282		if err != nil {
   283			return ADKLoopResult{}, fmt.Errorf("adk_agent_create: %w", err)
   284		}
   285		runner := adk.NewRunner(ctx, adk.RunnerConfig{Agent: agent, EnableStreaming: config.EnableStreaming})
   286		input := make([]adk.Message, 0, len(messages))
   287		for _, message := range messages {
   288			if message == nil {
   289				return ADKLoopResult{}, errors.New("adk_message_nil")
   290			}
   291			input = append(input, message)
   292		}
   293		iter := runner.Run(ctx, input)
   294		result := ADKLoopResult{Messages: []*schema.Message{}, ToolCalls: []schema.ToolCall{}, ToolResults: []*schema.Message{}}
   295		var lastAssistant *schema.Message
   296		seenToolCallIDs := make(map[string]struct{})
   297		for {
   298			event, ok := iter.Next()
   299			if !ok {
   300				break
   301			}
   302			if event == nil {
   303				continue
   304			}
   305			if event.Err != nil {
   306				return ADKLoopResult{}, fmt.Errorf("adk_run: %w", event.Err)
   307			}
   308			if event.Output == nil || event.Output.MessageOutput == nil {
   309				continue
   310			}
   311			message, getErr := event.Output.MessageOutput.GetMessage()
   312			if getErr != nil {
   313				return ADKLoopResult{}, fmt.Errorf("adk_message_output: %w", getErr)
   314			}
   315			if message == nil {
   316				continue
   317			}
   318			copyMessage := message
   319			result.Messages = append(result.Messages, copyMessage)
   320			if len(copyMessage.ToolCalls) > 0 {
   321				for _, call := range copyMessage.ToolCalls {
   322					if strings.TrimSpace(call.ID) != "" {
   323						if _, seen := seenToolCallIDs[call.ID]; seen {
   324							continue
   325						}
   326						seenToolCallIDs[call.ID] = struct{}{}
   327					}
   328					result.ToolCalls = append(result.ToolCalls, call)
   329				}
   330			}
   331			if copyMessage.Role == schema.Tool {
   332				result.ToolResults = append(result.ToolResults, copyMessage)
   333			}
   334			if copyMessage.Role == schema.Assistant {
   335				lastAssistant = copyMessage
   336			}
   337			result.Iterations++
   338		}
   339		// The final assistant event is authoritative. Never reuse an earlier text
   340		// candidate when ADK has emitted a later empty assistant termination; doing
   341		// so would publish text from before the last tool round (R-02).
   342		result.FinalMessage = lastAssistant
   343		if result.FinalMessage == nil {
   344			return ADKLoopResult{}, errors.New("adk_final_message_missing")
   345		}
   346		if strings.TrimSpace(result.FinalMessage.Content) == "" && len(result.FinalMessage.ToolCalls) == 0 {
   347			return ADKLoopResult{}, errors.New("adk_final_message_empty")
   348		}
   349		return result, nil
   350	}

   488		if trace == nil {
   489			return
   490		}
   491		allowed := make(map[string]struct{}, len(calls))

   496		}
   497		invocations := trace.Invocations[:0]

   499			if _, ok := allowed[invocation.CallID]; ok {

   501			}
