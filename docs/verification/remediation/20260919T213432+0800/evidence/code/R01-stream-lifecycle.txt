   183	// runProviderQueuedStream keeps the physical queue lease and cancellation
   184	// watcher alive until the returned StreamReader reaches EOF or is closed. A
   185	// plain runProviderQueued call cannot be used here: it would release its
   186	// defer-cancel immediately after receiving the reader, before the provider has
   187	// emitted any bytes.
   188	func runProviderQueuedStream(p *ProviderClient, ctx context.Context, role, scenario string, priority int, diagnosticID string, fn func(context.Context) (*schema.StreamReader[*schema.Message], error)) (*schema.StreamReader[*schema.Message], error) {
   189		if p == nil {
   190			return nil, errors.New("provider_unavailable")
   191		}
   192		if providerQueueBypassed(ctx) {
   193			return fn(ctx)
   194		}
   195		p.refreshQueueLimits(ctx)
   196		reader, writer := schema.Pipe[*schema.Message](1)
   197		runCtx, cancel := context.WithCancel(ctx)
   198		stopWatch := p.watchProviderCancellation(runCtx, cancel)
   199		go func() {
   200			defer stopWatch()
   201			defer cancel()
   202			releaseRedis, redisEnabled, redisErr := p.acquireProviderRedisSlot(runCtx, role, priority, p.queueFor(role).currentLimit(), diagnosticID)
   203			if redisErr != nil {
   204				if diagnosticID != "" {
   205					p.runtimeSupport().UpdateModelRunState(ctx, diagnosticID, providerRunFailed, redisErr)
   206				}
   207				writer.Send(nil, redisErr)
   208				writer.Close()
   209				return
   210			}
   211			if redisEnabled {
   212				defer releaseRedis()
   213			}
   214			err := p.queueFor(role).submit(runCtx, priority, func(taskCtx context.Context) error {
   215				if guard := providerExecutionGuard(taskCtx); guard != nil {
   216					if guardErr := guard(taskCtx); guardErr != nil {
   217						return guardErr
   218					}
   219				}
   220				inner, streamErr := fn(taskCtx)
   221				if streamErr != nil {
   222					return streamErr
   223				}
   224				if inner == nil {
   225					return errors.New("provider_stream_reader_missing")
   226				}
   227				defer inner.Close()
   228				innerDone := make(chan struct{})
   229				defer close(innerDone)
   230				go func() {
   231					select {
   232					case <-taskCtx.Done():
   233						inner.Close()
   234					case <-innerDone:
   235					}
   236				}()
   237				for {
   238					message, recvErr := inner.Recv()
   239					if errors.Is(recvErr, io.EOF) {
   240						return nil
   241					}
   242					if recvErr != nil {
   243						return recvErr
   244					}
   245					if writer.Send(message, nil) {
   246						inner.Close()
   247						return context.Canceled
   248					}
   249				}
   250			}, func(status string, runErr error) {
   251				if p.DB == nil || diagnosticID == "" {
   252					return
   253				}
   254				p.runtimeSupport().UpdateModelRunState(ctx, diagnosticID, status, runErr)
   255			})
   256			if err != nil && !errors.Is(err, context.Canceled) {
   257				writer.Send(nil, err)
   258			}
   259			writer.Close()
   260		}()
   261		return reader, nil
   262	}
