    10	// ModelTaskKind names the bounded responsibilities that may call a model.
    11	// A task owns its input contract, prompt fragments, provider invocation and
    12	// output normalization. Domain callers pass facts, not provider messages or
    13	// schemas.
    14	type ModelTaskKind string
    15	
    16	const (
    17		ModelTaskMainConversation     ModelTaskKind = "main_conversation"
    18		ModelTaskQueryContinuation    ModelTaskKind = "query_continuation"
    19		ModelTaskTakeoverJudge        ModelTaskKind = "takeover_judge"
    20		ModelTaskTakeoverReply        ModelTaskKind = "takeover_reply"
    21		ModelTaskStructuredAssessment ModelTaskKind = "structured_assessment"
    22		ModelTaskText                 ModelTaskKind = "text"
    23		ModelTaskMultimodalAssessment ModelTaskKind = "multimodal_assessment"
    24		ModelTaskEmbedding            ModelTaskKind = "embedding"
    25		ModelTaskStream               ModelTaskKind = "stream"
    26	)

    28	// ModelTask is stable metadata for diagnostics and assignment selection. It
    29	// deliberately contains no raw provider messages, schema or mutable domain
    30	// state. Concrete task input types below are the public boundary for callers.
    31	type ModelTask struct {
    32		Kind       ModelTaskKind
    33		Role       string
    34		Scenario   string
    35		SchemaName string
    36	}

    44	func (a *App) modelTaskProvider() (*ProviderClient, error) {
    45		if a == nil || a.Provider == nil {
    46			return nil, errors.New("model_task_provider_unavailable")
    47		}
    48		return a.Provider, nil
    49	}

    58	func (a *App) RunInitializationTask(ctx context.Context, input InitializationTaskInput) (map[string]any, error) {
    59		if strings.TrimSpace(input.Description) == "" {
    60			return nil, errors.New("initialization_description_required")
    61		}
    62		provider, err := a.modelTaskProvider()
    63		if err != nil {
    64			return nil, err
    65		}
    66		messages := initializationAnalysisMessages(input.Description)
    67		return provider.Structured(WithProviderScenario(ctx, "initialization"), "initialization", messages)
    68	}

   195	const scheduleGenerationTaskInstruction = "Return one compact object with items and reschedule_policy. items must contain 8-16 objects covering the complete local day contiguously from 00:00 through the next 00:00 in the supplied timezone. Every item needs start_at, end_at, activity, scene, location, item_type, status, priority, flexibility, interruption_cost. Keep activity, scene, and location each under 80 Chinese characters; use one concrete activity and one concrete scene per item, never combine alternatives with '/', '／', '、', or '或'. Merge adjacent periods with the same activity and scene instead of producing many small segments. priority, flexibility, and interruption_cost are normalized numbers from 0 to 1 (never a 1-10 score). Use RFC3339 timestamps with the supplied timezone. Do not return markdown or foundation fields."

   197	func (a *App) RunScheduleGenerationTask(ctx context.Context, input ScheduleGenerationTaskInput) (map[string]any, error) {

   212	// Projection-backed tasks own selection of context surfaces, operation rules,
   213	// tool catalog and schema. They return the assembled projection because the
   214	// caller still owns domain validation and settlement after model execution.
   215	type NativeCognitionTaskInput struct {
   216		EventType  string
   217		Fact       []byte
   218		Projection ContextProjection
   219	}
   220	
   221	func (a *App) RunNativeCognitionTask(ctx context.Context, input NativeCognitionTaskInput) (ProjectionTaskResult, error) {
   222		definitions := capabilityCatalog(a.capabilityRegistry(), CapabilitySurfaceNativeCognition)
   223		schema := nativeCognitionResponseSchema()
   224		assembly, projection, err := a.assembleProjectionPromptForSurface(ctx, ProviderContextSurfaceNativeCognition, input.Projection, "cognitive_assessment", []string{providerContextAuthorityRule, nativeCognitionInstruction}, jsonString(map[string]any{"event_type": input.EventType, "fact": compactProviderFact(input.Fact)}), definitions, "native_cognition_response", schema)
   225		if err != nil {
   226			return ProjectionTaskResult{}, err
   227		}
   228		provider, err := a.modelTaskProvider()
   229		if err != nil {
   230			return ProjectionTaskResult{}, err
   231		}
   232		providerCtx := WithPromptDiagnostics(WithProviderScenario(ctx, "native_cognition"), assembly.Diagnostics)
   233		completion, err := provider.StructuredAssembledWithToolsSchema(providerCtx, "cognitive_assessment", assembly.Messages, definitions, "native_cognition_response", schema, true)
   234		return ProjectionTaskResult{Completion: completion, Projection: projection, Diagnostics: assembly.Diagnostics}, err
   235	}

   338	func validateModelTask(task ModelTask) error {
