     8	// ProviderRuntimeSupport is the narrow persistence/diagnostic seam used by a
     9	// physical model call. It deliberately does not expose App, Repository,
    10	// CapabilityRuntime or domain state; the composition root supplies it when a
    11	// ProviderClient is created.
    12	type ProviderRuntimeSupport interface {
    13		RecordModelRun(context.Context, string, string, string, string, any, any, string, string)
    14		PersistModelRunLifecycle(context.Context, string, string, string, string, string, int, any, any, string, string) (string, error)
    15		RecordQueuedModelRun(context.Context, string, string, string, string, string, int, any) string
    16		UpdateModelRunState(context.Context, string, string, error)
    17		UpdateModelRunPromptMetrics(context.Context, string, map[string]any, time.Duration)
    18		RecordDiagnosticEvent(context.Context, string, string, string, string, string, any)
    19		RecordLifecycleDiagnostic(context.Context, LifecycleDiagnostic) (string, error)
    20		RecordLifecycleDiagnosticBestEffort(context.Context, LifecycleDiagnostic)
    21	}
    22	
    23	type providerRuntimeSupport struct{ DB *PostgresRepository }
    24	
    25	func newProviderRuntimeSupport(db *PostgresRepository) ProviderRuntimeSupport {
    26		return providerRuntimeSupport{DB: db}
    27	}
