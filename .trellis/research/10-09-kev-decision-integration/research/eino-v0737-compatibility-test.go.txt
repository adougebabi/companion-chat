type visibleState struct { mu sync.Mutex; loaded bool; executions int }
type visibleModel struct { inner model.ToolCallingChatModel; state *visibleState }
func (m *visibleModel) WithTools(infos []*schema.ToolInfo)(model.ToolCallingChatModel,error){ b,e:=m.inner.WithTools(infos); return &visibleModel{b,m.state},e }
func (m *visibleModel) options(opts []model.Option) []model.Option { m.state.mu.Lock(); loaded:=m.state.loaded; m.state.mu.Unlock(); infos:=[]*schema.ToolInfo{{Name:"discover",Desc:"Load a permitted capability"}}; if loaded { infos=append(infos,&schema.ToolInfo{Name:"hidden",Desc:"Return actual result"}) }; return append(append([]model.Option{},opts...),model.WithTools(infos)) }
func(m *visibleModel)Generate(ctx context.Context,input []*schema.Message,opts ...model.Option)(*schema.Message,error){return m.inner.Generate(ctx,input,m.options(opts)...)}
func(m *visibleModel)Stream(ctx context.Context,input []*schema.Message,opts ...model.Option)(*schema.StreamReader[*schema.Message],error){return m.inner.Stream(ctx,input,m.options(opts)...)}
type testTool struct{name string; state *visibleState}
func(t *testTool)Info(context.Context)(*schema.ToolInfo,error){return &schema.ToolInfo{Name:t.name,Desc:t.name},nil}
func(t *testTool)InvokableRun(ctx context.Context,arg string,opts ...tool.Option)(string,error){ t.state.mu.Lock();defer t.state.mu.Unlock();t.state.executions++;if t.name=="discover"{t.state.loaded=true;return `{"loaded":"hidden"}`,nil};return `{"actual":"hidden-result"}`,nil }
func TestNativeLoopCallTimeToolsGrow(t *testing.T){ runNative(t,false) }
func TestNativeBatchGateBlocksAllTools(t *testing.T){runNative(t,true)}
func runNative(t *testing.T,block bool){
 state:=&visibleState{}; requests:=[][]string{}; var requestMu sync.Mutex
 server:=httptest.NewServer(http.HandlerFunc(func(w http.ResponseWriter,r *http.Request){
 var body map[string]any; if e:=json.NewDecoder(r.Body).Decode(&body);e!=nil{t.Error(e);w.WriteHeader(400);return}; names:=[]string{}; for _,v:=range body["tools"].([]any){names=append(names,v.(map[string]any)["function"].(map[string]any)["name"].(string))};requestMu.Lock(); requests=append(requests,names);n:=len(requests);requestMu.Unlock()
 response:=`{"id":"response","object":"chat.completion","model":"test","choices":[{"index":0,"message":{"role":"assistant","content":"done"},"finish_reason":"stop"}]}`
 if n<=2 { name:="discover";if n==2{name="hidden"}; response=fmt.Sprintf(`{"id":"response","object":"chat.completion","model":"test","choices":[{"index":0,"message":{"role":"assistant","tool_calls":[{"id":"call_%d","type":"function","function":{"name":"%s","arguments":"{}"}}]},"finish_reason":"tool_calls"}]}`,n,name) }
 w.Header().Set("Content-Type","application/json");fmt.Fprint(w,response)
 }));defer server.Close()
 chat,e:=openai.NewChatModel(context.Background(),&openai.ChatModelConfig{APIKey:"test",BaseURL:server.URL+"/v1",Model:"test",HTTPClient:server.Client()});if e!=nil{t.Fatal(e)}
 config:=&adk.ChatModelAgentConfig{Name:"compat",Description:"compat",Model:&visibleModel{chat,state},MaxIterations:4,ToolsConfig:adk.ToolsConfig{ToolsNodeConfig:compose.ToolsNodeConfig{Tools: nil,ExecuteSequentially:true}}}
 config.ToolsConfig.Tools=[]tool.BaseTool{&testTool{"discover",state},&testTool{"hidden",state}}
 if block { config.Middlewares=[]adk.AgentMiddleware{{AfterChatModel:func(ctx context.Context,s *adk.ChatModelAgentState)error{if len(s.Messages)>0&&len(s.Messages[len(s.Messages)-1].ToolCalls)>0{return fmt.Errorf("gate_blocked")};return nil}}} }
 agent,e:=adk.NewChatModelAgent(context.Background(),config);if e!=nil{t.Fatal(e)}
 runner:=adk.NewRunner(context.Background(),adk.RunnerConfig{Agent:agent})
 iter:=runner.Run(context.Background(),[]adk.Message{schema.UserMessage("discover then call hidden")});hadError:=false
 for {ev,ok:=iter.Next();if !ok{break};if ev.Err!=nil{hadError=true}}
 if block{if !hadError||state.executions!=0{t.Fatalf("gate error=%v executions=%d",hadError,state.executions)};return}
 if hadError||state.executions!=2||len(requests)!=3{t.Fatalf("error=%v executions=%d requests=%v",hadError,state.executions,requests)}
 if len(requests[0])!=1||requests[0][0]!="discover"||len(requests[1])!=2||requests[1][1]!="hidden"{t.Fatalf("schema did not grow: %v",requests)}
 t.Logf("native loop wire schema=%v; real tool executions=%d",requests,state.executions)
}
